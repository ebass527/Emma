<script>
let lines = document.querySelectorAll('.line');
let currentLine = 0;
let autoTimeout;
let tapTimes = [];
let bpm = 0;
let minInterval = 200;

function highlightLine(index) {
  lines.forEach(line => line.classList.remove('highlight'));
  if (lines[index]) lines[index].classList.add('highlight');
}

function scrollLine(index) {
  if (lines[index]) lines[index].scrollIntoView({behavior: 'smooth', block: 'center'});
}

function startAuto() {
  pauseAuto();
  if (bpm > 0) {
    highlightLine(currentLine);
    scrollLine(currentLine);
    scheduleNextLine();
  } else {
    alert("Tap a few times to set tempo first!");
  }
}

function pauseAuto() {
  clearTimeout(autoTimeout);
}

function scheduleNextLine() {
  if (currentLine >= lines.length - 1) return;
  const interval = 60000 / bpm;
  autoTimeout = setTimeout(() => {
    currentLine++;
    highlightLine(currentLine);
    scrollLine(currentLine);
    scheduleNextLine();
  }, interval);
}

function tapTempo() {
  const now = Date.now();
  if (tapTimes.length && now - tapTimes[tapTimes.length - 1] < minInterval) return;
  tapTimes.push(now);
  if (tapTimes.length > 6) tapTimes.shift();
  if (tapTimes.length >= 2) {
    let intervals = [];
    for (let i = 1; i < tapTimes.length; i++) intervals.push(tapTimes[i] - tapTimes[i-1]);
    let avgInterval = intervals.reduce((a,b)=>a+b,0)/intervals.length;
    bpm = 60000 / avgInterval;
    document.getElementById('tempoDisplay').innerText = `Tempo: ${Math.round(bpm)} BPM`;
  }
}

document.getElementById('tapButton').addEventListener('click', tapTempo);
document.body.addEventListener('keydown', (e) => {
  if (e.code === "Space") tapTempo();
});
</script>
